Particle Physics & Engine Code

The Particle Cursor Trail works by instantiating lightweight JS objects at the mouse context (x, y) on every frame. Each particle updates its position vector using velocity, drag, and gravity while decreasing its lifespan to shrink and fade opacity. 
class Particle {
  constructor(x, y, options) {
    this.x = x;
    this.y = y;
    this.size = options.baseSize * (0.8 + Math.random() * 0.4);
    
    // Radial Velocity Vector calculation
    const angle = Math.random() * Math.PI * 2;
    const speed = Math.random() * options.speedSpread;
    this.vx = Math.cos(angle) * speed;
    this.vy = Math.sin(angle) * speed;

    this.life = options.maxLife;
    this.maxLife = options.maxLife;
    this.color = options.palette.getColor(x, y);
  }

  update(gravity, turbulence, shrink) {
    // Add physics forces
    this.vx += (Math.random() - 0.5) * turbulence;
    this.vy += gravity + (Math.random() - 0.5) * turbulence;
    
    // Position integration
    this.x += this.vx;
    this.y += this.vy;

    // Lifespan & Size Decay
    this.life--;
    if (shrink) {
      this.currentSize = Math.max(0, (this.life / this.maxLife) * this.size);
    }
  }

  draw(ctx, blendMode) {
    const alpha = this.life / this.maxLife;
    ctx.globalCompositeOperation = blendMode;
    ctx.fillStyle = this.color.replace('ALPHA', alpha);
    
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.currentSize, 0, Math.PI * 2);
    ctx.fill();
  }
}
